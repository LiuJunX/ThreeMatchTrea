✓ 已完成P1 P5 P8
interface IObstacleSystem { // ① 直接命中（CellEliminator 调用） ObstacleHitResult TryHit(ref state, pos, ctx, ...); // ② 邻接反应（MatchProcessor 每组消除后调用） void NotifyBatchElimination(ref state, tiles[], ...); // ③ 全局反应（MatchProcessor 每组匹配颜色后调用） void NotifyGlobalColorElimination(ref state, color, ...); }
